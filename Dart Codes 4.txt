class Animal{
  void makeSound(){
    print('Animal makes sound');
  }
  void view(){
    print('View');
  }
}
class Dog extends Animal{
  @override
  void view(){
    print('Dog Barks');
  }
}

void main(){
  Animal a1 = Dog();
  a1.view();
  a1.makeSound();
}

class Animal{
  void makeSound(){
    print('Animal makes sound');
  }
  void view(){
    print('View');
  }
}
class Dog extends Animal{
  @override
  void view(){
    print('Dog Barks');
  }
  void info(){
    print('This is polymorphism example');
  }
}

void main(){
  Animal a1 = Dog();
  a1.view();
  a1.makeSound();
  Dog d1 = Dog();
  d1.info();
}

class Animal {
  void sum(){}
  void view(){
    print('View');
  }
  void makeSound(){
    print('Animal makes sound');
  }
}
class Dog implements Animal{
  void sum(){
    print('Result is sum');
  }
  void view(){
    print('View');
  }
  void makeSound(){
    print('Animal makes sound');
  }
}

void main(){
  Dog d1 = Dog();
  d1.sum();
  d1.view();
  d1.makeSound();
}
